\begin{algorithm}
	\caption{Build a Huffman coding tree}
	
	\begin{algorithmic}[1]
		\Require $freq$ : an array of frequency of each value $v$.
		\Ensure a Huffman coding tree.
		
		\Function{BuildHuffmanTree}{$freq$}   
			\ForAll{value $v$}
				\State Create a node $n$ with frequency $freq[v]$ and value $v$.
				\State Put $n$ in an array $nodes$.
			\EndFor
			\\
			\While{there are more than 1 node in $nodes$}
				\State Find $n1$ in $nodes$ with the least frequency.
				\State Remove $n1$ from $nodes$.
				\\
				\State Find $n2$ in $nodes$ with the least frequency.
				\State Remove $n2$ from $nodes$.
				\\
				\State Create a node $n3$ with 
					\State\hspace{\algorithmicindent} $frequency \leftarrow n1.frequency + n2.frequency$
					\State\hspace{\algorithmicindent} $left\_child \leftarrow n1$.
					\State\hspace{\algorithmicindent} $right\_child \leftarrow n2$.
				\\
				\State Put $n3$ back to $nodes$.
			\EndWhile
			\\
			\State \Return the only node in $nodes$.
		\EndFunction
	\end{algorithmic}
\end{algorithm}



\begin{algorithm}
	\caption{Serialize a Huffman coding tree}
	
	\begin{algorithmic}[1]
		\Require $node$ : the root node of a Huffman coding tree.
		\Ensure $bit\_topology$ : a bit sequence represents the topology of the tree.
		\Ensure $bit\_value$ : a bit sequence represents every leaf value $v$ of the tree.
		
		\Procedure{SerializeHuffmanTree}{$node$}   
			\If{$node$ is empty}
				\State Do nothing.
			\ElsIf{$node$ is a leaf}
				\State Append a $0$ at the end of $bit\_topology$.
				\State Append the value $v$ of $node$ at the end of $bit\_value$ bit by bit.
			\Else
				\State Append a $1$ at the end of $bit\_topology$.
				\State \Call{SerializeHuffmanTree}{$node.left\_child$}.
				\State \Call{SerializeHuffmanTree}{$node.right\_child$}.
			\EndIf
		\EndProcedure
	\end{algorithmic}
\end{algorithm}

\begin{algorithm}
	\caption{Unserialize a Huffman coding tree}
	
	\begin{algorithmic}[1]
		\Require $bit\_topology$ : a bit sequence represents the topology of the tree.
		\Require $bit\_value$ : a bit sequence represents every leaf value $v$ of the tree.
		\Require $size\_value$ : the size of bits of each value $v$.
		\Ensure a Huffman coding tree.
		
		\Function{UnserializeHuffmanTree}{$bit\_topology, bit\_value$}   
			\State Read the first bit $b$ from $bit\_topology$.
			\State Remove $b$ from $bit\_topology$.
			\\
			\If{$b$ is 0}
				\State Restore value $v$ from the first $size\_value$ bits of $bit\_value$.
				\State Remove the first $size\_value$ bits from $bit\_value$.
				\State Create a node $n$ with $v$.
			\Else
				\State Acquire $n1 \gets$ \Call{UnserializeHuffmanTree}{$bit\_topology, bit\_value$}.
				\State Acquire $n2 \gets$ \Call{UnserializeHuffmanTree}{$bit\_topology, bit\_value$}.
				\State Create a node $n$ with
					\State\hspace{\algorithmicindent} $left\_child \gets n1$.
					\State\hspace{\algorithmicindent} $right\_child \gets n2$.
			\EndIf
			\\
			\State \Return $n$.
		\EndFunction
	\end{algorithmic}
\end{algorithm}

\begin{algorithm}
	\caption{Assign values to Huffman coding rule}
	\label{alg:msb:assignhr}
	
	\begin{algorithmic}[1]
		\Require a label map $T$ of size $w \times h$.
		\Require a fixed Huffman coding tree $H$ without value.
		\Ensure a Huffman coding tree $H$ assigned with values $v$.
		
		\Function{AssignHuffmanValue}{$T, H$}  
		 
			\State $F \gets \{f_n \mid n \in [0, 8]\}$.
			\\
			\ForAll{$\{t_{i, j} \in T \mid i \in [1, h), j \in [1, w)\}$}
				\State Count the frequency $f_{t_{i, j}}$.
			\EndFor
			\\
			\State Sort $V = \{v \in [0, 8]\}$ by $f_v$.
			\State Assign $V$ to $H$.
			\\
			\State \Return $H$.
		\EndFunction
	\end{algorithmic}
\end{algorithm}

\begin{algorithm}
	\caption{Generate encrypted image}
	\label{alg:msb:genenc}
	
	\begin{algorithmic}[1]
		\Require a cover image $I$ of size $w \times h$.
		\Require a label map $T$.
		\Require a key stream $ks$.
		\Require a Huffman coding tree $H$.
		\Ensure an encrypted image $I_e$.
		\Ensure a coordinate $(c_x, c_y)$ in $I_e$ to start information hiding.
		
		\Function{EncryptImage}{$I, T, ks, H$}  
			
			\State $I_e \gets I \oplus ks$
			\\
			\ForAll{$\{t_{i, j} \in T \mid i \in [1, h), j \in [1, w)\}$}
				\State $code \gets Encode(t_{i, j}, H)$.
				\State $EncodedLabelMap \gets EncodedLabelMap \Vert code$.
				\State $len \gets len + \left \vert code \right \vert$
			\EndFor
			\\
			\State Allocate a bit array $bs$ with
				\State\hspace{\algorithmicindent} 32 bits for $\{Encode(v, H) \mid v \in [0, 8] \}$.
				\State\hspace{\algorithmicindent} 20 bits to store $len$.
				\State\hspace{\algorithmicindent} $len$ bits to store $EncodedLabelMap$.
				\State\hspace{\algorithmicindent} $w + h - 1$ bytes to store the first row and the first col of $I_e$.
			\\
			\State Replace the first row of $I_e$ with $w$ bytes of $bs$.
			\State Replace the first col of $I_e$ with $h - 1$ bytes of $bs$.
			\\
			\ForAll{$ i \in [1, h)$ and $ j \in [1, w)$}
				\If{there is no more $bs$}
					\State $(c_x, c_y) \gets (i, j)$
					\State Leave the loop.
				\Else
					\State Replace $t_{i, j}$ bits MSB of $I_{e(i, j)}$ with $bs$. 
				\EndIf
			\EndFor
			\\
			\State \Return $I_e$ and $(c_x, c_y)$.
		
		\EndFunction
	\end{algorithmic}
\end{algorithm}

\begin{algorithm}
	\caption{Embed secret bits}
	\label{alg:msb:embed}
	
	\begin{algorithmic}[1]
		\Require a label map $T$.
		\Require a location $(c_x, c_y)$ to start information hiding.
		\Require an encrypted image $I_e$ of $w \times h$.
		\Require secret bits $sb$.
		\Ensure an encrypted image $I_e$ embedded with $sb$.
		
		\Function{EmbedSecret}{$T, (c_x, c_y), I_e, sb$}  
			\ForAll{$\{t_{i, j} \in T \mid (i = c_x \wedge j \in [c_y, w)) \vee (i \in (c_x, h), j \in [1, w))\}$}
				\If{there is no more $sb$}
					\State Leave the loop.
				\Else
					\State Replace $t_{i, j}$ bits MSB of $I_{e(i, j)}$ with $sb$. 
				\EndIf
			\EndFor
		
			\State \Return $I_e$.
		
		\EndFunction
	\end{algorithmic}
\end{algorithm}
