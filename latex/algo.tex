\begin{algorithm}
	\caption{Build a Huffman coding tree}
	
	\begin{algorithmic}[1]
		\Require $freq$ : an array of frequency of each value $v$.
		\Ensure a Huffman coding tree.
		\Function{BuildHuffmanTree}{$freq$}   
		
			\ForAll{$freq[v]$ in $freq$}
				\State Create a node $n$ with frequency $freq[v]$ and value $v$.
				\State Put $n$ in an array $nodes$.
			\EndFor
		
			\While{there are more than 1 element in $nodes$}
				\State Find $n1$ in $nodes$ with the least frequency.
				\State Remove $n1$ from $nodes$.
				
				\State Find $n2$ in $nodes$ with the least frequency.
				\State Remove $n2$ from $nodes$.
				
				\State Create a node $n3$ with 
				\State\hspace{\algorithmicindent} $frequency \gets n1.frequency + n2.frequency$
				\State\hspace{\algorithmicindent} $value \gets none$.
				\State\hspace{\algorithmicindent} $left\_child \gets n1$.
				\State\hspace{\algorithmicindent} $right\_child \gets n2$.
					
				
				\State Put $n3$ back to $nodes$.
			\EndWhile
			\State \Return the last element in $nodes$.
		\EndFunction
	\end{algorithmic}
\end{algorithm}



\begin{algorithm}
	\caption{Serialize a Huffman coding tree}
	
	\begin{algorithmic}[1]
		\Require $node$ : the root node of a Huffman coding tree.
		\Ensure $bit\_topology$ : a bit sequence represents the topology of the tree.
		\Ensure $bit\_value$ : a bit sequence represents every leaf value $v$ of the tree.
		\Procedure{SerializeHuffmanTree}{$node$}   
		
		\If{$node$ is empty}
			\State \Return nothing.
		\EndIf
		
		\If{$node$ is a leaf}
			\State Append a $0$ at the end of $bit\_topology$.
			\State Append the value $v$ of $node$ at the end of $bit\_value$ bit by bit.
		\Else
			\State Append a $1$ at the end of $bit\_topology$.
			\State \Call{SerializeHuffmanTree}{$node.left\_child$}.
			\State \Call{SerializeHuffmanTree}{$node.right\_child$}.
		\EndIf
		\EndProcedure
	\end{algorithmic}
\end{algorithm}

\begin{algorithm}
	\caption{Unserialize a Huffman coding tree}
	
	\begin{algorithmic}[1]
		\Require $bit\_topology$ : a bit sequence represents the topology of the tree.
		\Require $bit\_value$ : a bit sequence represents every leaf value $v$ of the tree.
		\Require $size\_value$ : the size of bits of each value $v$.
		\Ensure a Huffman coding tree.
		\Function{UnserializeHuffmanTree}{$bit\_topology, bit\_value$}   
		
		\State Read the first bit $b$ from $bit\_topology$.
		\State Remove $b$ from $bit\_topology$.
		
		\If{$b$ is 0}
			\State Restore value $v$ from the first $size\_value$ bits of $bit\_value$.
			\State Remove the first $size\_value$ bits from $bit\_value$.
			\State Create a node $n$ with $v$.
		\Else
			\State Acquire $n1 \gets$ \Call{UnserializeHuffmanTree}{$bit\_topology, bit\_value$}.
			\State Acquire $n2 \gets$ \Call{UnserializeHuffmanTree}{$bit\_topology, bit\_value$}.
			\State Create a node $n$ with
				\State\hspace{\algorithmicindent} $left\_child \gets n1$.
				\State\hspace{\algorithmicindent} $right\_child \gets n2$.
		\EndIf
		\State \Return $n$.
		\EndFunction
	\end{algorithmic}
\end{algorithm}
