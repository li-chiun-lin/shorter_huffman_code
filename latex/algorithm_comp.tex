The algorithm of Reversible Data Hiding in Encrypted Images Based on Multi-MSB Prediction and Huffman Coding, a.k.a. msb\_huffman. 


Our algorithm, a.k.a. shorter\_huffman.

   
\section{msb\_huffman embedding}
    \begin{enumerate}
        \item Generate Label Map. 
        \item[] INPUT: Cover image
        \item[] OUTPUT: Label map, frequency table
          		\begin{enumerate}
          			\item Skip first row and first column since they are defined as Reference Values.
          			\item For the rest of the pixels of cover image
          			\begin{enumerate}
          				\item 
           				$pv(i, j) = \left\{ 
           				\begin{array}{rl}
           					max(x_2, x_3)   &, x_1 \leq min(x_2, x_3)\\
           					min(x_2, x_3)   &, x_1 \geq max(x_2, x_3)\\
           					x_2 + x_3 - x_1 &, otherwise
           				\end{array} \right. $ 
          				\item $idx = \textrm{the index of the MSB of } pv(i, j) \oplus pixel(i, j) \textrm{ counting from LSB.}$
          				\item 
           				$t = \left\{ 
           				\begin{array}{rl}
           					8   &, idx = 0\\
           					8 - idx + 1   &, otherwise
           				\end{array} \right.$
           				\item calculate $freq[t]$.
          			\end{enumerate}
          		\end{enumerate}
        \item Generate Huffman coding rule from a fixed Huffman tree.
        \item[] INPUT: frequency table
        \item[] OUTPUT: Huffman coding rule, capacity
           	\begin{enumerate}
           		\item Skip first row and first column.
           		\item For the rest of the label map, calculate the $freq$ of every $t$.
           		\item Sort $t$ by $freq$.
           		\item Map each $t$ to a fixed Huffman coding rule \{00, 01, 100, 101, 1100, 1101, 1110, 11110, 11111\}.
           		\item For each $t$ in frequency table, calculate
           			\begin{enumerate}
           				\item $LM = \sum_{t=0}^{8} (freq[t] * \textrm{the Huffman code length of }t) $
           				\item $cap = \sum_{t=0}^{8} (freq[t] * c[t]), c[t] = \left\{ 
           				\begin{array}{rl}
           					t      &, t = 8 \\
           					t + 1  &, otherwise
           				\end{array} \right. $
           			\end{enumerate}
           		\item the total $capacity$ is $cap - LM - header$, the $header$ is described in next part.
       		\end{enumerate} 
       	
       	\item Encode label map
       	\item[] INPUT: Label map, Huffman coding rule
       	\item[] OUTPUT: encoded label map.
       	
       	\item Encrypt the cover image.
       	\item[] INPUT: Cover image, key stream
       	\item[] OUTPUT: Encrypted image
       	
       	\item Generate Side information
       	\item[] INPUT: Label map, Huffman coding rule, encrypted reference bytes
       	\item[] OUTPUT: Side information
       		\begin{enumerate}
       			\item the first 32 bits are the Huffman coding rule sorted by $t$.
       			\item the following 20 bits are the value of $LM$.
       			\item the $LM$ bits Huffman encoded label map follows.
       			\item the encrypted first row and first column reference values are appended afterward.
       		\end{enumerate}
       	
       	\item Replace the encrypted reference byte with side information(part 1).
       	\item[] INPUT: encrypted image, side information
       	\item[] OUTPUT: stego image(part 1)
       	
       	\item The rest of the side information(part 2) is embed into the $t+1$ MSB of each encrypted pixel.
       	\item[] INPUT: stego image(part 1)
       	\item[] OUTPUT: stego image(part 2)
       	
       	\item the rest of the $capacity$ is used to embed secret bits.
       	\item[] INPUT: stego image(part 2)
       	\item[] OUTPUT: stego image
	\end{enumerate} 

\section{msb\_huffman extracting}
	\begin{enumerate}
		\item Extract side information(part 1) from the first row and first column of the stego image.
		\item[] INPUT: stego image
		\item[] OUTPUT: side information(part 1), Huffman coding rule, label map(part 1)
		\begin{enumerate}
			\item Extract the Huffman coding rules from the first 32 bits of the extracted side information.
			\item Extract $LM$ from the following 20 bits.
			\item For the rest of the extracted side information(part 1), decode by Huffman coding rules and re-construct as many $t$ of label map as possible.
		\end{enumerate}
		
		\item According to the already re-constructed part of the label map, keep extract side information(part 2).
		\item[] INPUT: stego image, Huffman coding rule, label map(part 1)
		\item[] OUTPUT: side information(part 2), label map.
		\item[] For every encrypted pixel except the reference bytes, calculate
			\begin{enumerate}
				\item Extract the $t+1$ MSB from the encrypted $pixel(i, j)$ and put them in a $buffer$.
					\begin{enumerate}
						\item if the $buffer$ become a valid Huffman code, then decode it into the next $t$.
					\end{enumerate}
				\item This process stops when all the $t$ of the lable map has been decoded.
			\end{enumerate}
		\item Extract the encrypted reference values and restore them into the stego image.
		\item[] INPUT: stego image, label map
		\item[] OUTPUT: encrypted image(part 1).
		
		\item Extract secret bits.
		\item[] INPUT: stego image, label map
		\item[] OUTPUT: secret bits.
		
		\item Restore the encrypted image.
		\item[] INPUT: encrypted image(part 1), label map
		\item[] OUTPUT: Encrypted image.
		\item[] For every encrypted pixel except the reference bytes, calculate
			\begin{enumerate}
				\item 
				$pv(i, j) = \left\{ 
				\begin{array}{rl}
					max(x_2, x_3)   &, x_1 \leq min(x_2, x_3)\\
					min(x_2, x_3)   &, x_1 \geq max(x_2, x_3)\\
					x_2 + x_3 - x_1 &, otherwise
				\end{array} \right. $ 
				\item replace the $t+1$ MSB of encrypted $pixel(i, j)$ with the $t+1$ MSB of $pv(i, j)$
			\end{enumerate}
		
		\item Decrypt the cover image.
		\item[] INPUT: Encrypted image, key stream
		\item[] OUTPUT: Cover image
		
	\end{enumerate}
        	
        
      



        


\section{shorter\_huffman embedding}
\begin{enumerate}
	\item Generate Label Map. 
	\item[] INPUT: Cover image
	\item[] OUTPUT: Label map, frequency table
	\begin{enumerate}
		\item Skip first 16 bytes since they are defined as Reference Values.
		\item For the rest of the pixels of the first row, $pv(0, j) = pixel(0, j - 1) \oplus pixel(0, j). $ 
		\item For the first pixel of each row, except the first pixel of the first row, $pv(i, 0) = pixel(i - 1, 0) \oplus pixel(i, 0). $ 
		\item For the rest of the pixels of cover image, 	$pv(i, j) = \left\{ 
		\begin{array}{rl}
			max(x_2, x_3)   &, x_1 \leq min(x_2, x_3)\\
			min(x_2, x_3)   &, x_1 \geq max(x_2, x_3)\\
			x_2 + x_3 - x_1 &, otherwise
		\end{array} \right. $
		\item For all the pixels except reference values,
		\begin{enumerate}
			\item $idx = \textrm{the index of the MSB of } pv(i, j) \oplus pixel(i, j) \textrm{ counting from LSB.}$
			\item 
			$t = \left\{ 
			\begin{array}{rl}
				8   &, idx = 0\\
				8 - idx + 1   &, otherwise
			\end{array} \right.$
			\item calculate $freq[t]$.
		\end{enumerate}
	\end{enumerate}
	\item Generate Huffman tree.
	\item[] INPUT: Label map, frequency table
	\item[] OUTPUT: Huffman coding rule, capacity
		\begin{enumerate}
			\item Generate a Huffman tree node for each $t$ and $freq[t]$ pair.
			\item while there are more then 1 such Huffman tree node.
			\begin{enumerate}
				\item find the 2 Huffman tree node with the lowest frequency.
				\item Generate a new Huffman tree node and set the 2 tree nodes as the left and right sub tree.
				\item set the frequency of the new tree node as the sum of the 2 tree nodes.
				\item remove the 2 tree nodes.
			\end{enumerate}
			\item map each $t$ with the corresponding Huffman code.
			\item For each $t$ in frequency table, calculate
			\begin{enumerate}
				\item $LM = \sum_{t=0}^{8} (freq[t] * \textrm{the Huffman code length of }t) $
				\item $cap = \sum_{t=0}^{8} (freq[t] * c[t]), c[t] = \left\{ 
				\begin{array}{rl}
					t      &, t = 8 \\
					t + 1  &, otherwise
				\end{array} \right. $
			\end{enumerate}
			\item the total $capacity$ is $cap - LM - header$, the $header$ is described in next part.
		\end{enumerate} 
	
	\item Encode label map.
	\item[] INPUT: Label map, Huffman coding rule
	\item[] OUTPUT: encoded label map
		
	\item Encrypt the cover image.
	\item[] INPUT: Cover image, key stream
	\item[] OUTPUT: Encrypted image
	
	\item Generate Side information
	\item[] INPUT: Label map, Huffman coding rule, encrypted reference bytes
	\item[] OUTPUT: Side information
	\begin{enumerate}
		\item the first 17 bits are the Huffman coding tree topology.
		\item the following 36 bits are the Huffman coding value.
		\item the following 20 bits are the value of $LM$.
		\item the $LM$ bits Huffman encoded label map follows.
		\item the first 16 encrypted reference bytes are appended afterward.
	\end{enumerate}
	
	\item Replace the 16 encrypted reference byte with side information(part 1).
	\item[] INPUT: encrypted image, side information
	\item[] OUTPUT: stego image(part 1)
	
	\item The rest of the side information(part 2) is embed into the $t+1$ MSB of each encrypted pixel.
	\item[] INPUT: stego image(part 1)
	\item[] OUTPUT: stego image(part 2)
	
	\item the rest of the $capacity$ is used to embed secret bits.
	\item[] INPUT: stego image(part 2)
	\item[] OUTPUT: stego image
\end{enumerate} 

\section{shorter\_huffman extracting}
\begin{enumerate}
	\item Extract side information(part 1) from the first 16 bytes of the stego image.
	\item[] INPUT: stego image
	\item[] OUTPUT: side information(part 1), Huffman coding rule, label map(part 1)
	\begin{enumerate}
		\item Extract the topology of Huffman coding tree from the first 17 bits of the extracted side information.
		\item Extract the following 36 bits as 9 Huffman coding values, 4 bits each.
		\item Extract $LM$ from the following 20 bits.
		\item For the rest of the extracted side inforation(part 1), decode by Huffman coding rules and re-construct as many $t$ of label map as possible.
	\end{enumerate}
	
	\item According to the already re-constructed part of the label map, keep extract side information(part 2).
	\item[] INPUT: stego image, Huffman coding rule, label map(part 1)
	\item[] OUTPUT: side information(part 2), label map.
	\item[] For every encrypted pixel except the reference bytes, calculate
	\begin{enumerate}
		\item Extract the $t+1$ MSB from the encrypted $pixel(i, j)$ and put them in a $buffer$.
		\begin{enumerate}
			\item if the $buffer$ become a valid Huffman code, then decode it into the next $t$.
		\end{enumerate}
		\item This process stops when all the $t$ of the lable map has been decoded.
	\end{enumerate}
	\item Extract the encrypted reference values and restore them into the stego image.
	\item[] INPUT: stego image, label map
	\item[] OUTPUT: encypted image(part 1).
	
	\item Extract secret bits.
	\item[] INPUT: stego image, label map
	\item[] OUTPUT: secret bits.
	
	\item Restore the encrypted image.
	\item[] INPUT: encrypted image(part 1), label map
	\item[] OUTPUT: Encrypted image.
	\item[] For every encrypted pixel except the reference bytes, calculate
	\begin{enumerate}
		\item for the pixel in the first row except reference bytes, $pv(0, j) = pixel(0, j - 1) \oplus pixel(0, j). $ 
		\item For the first pixel of each row, except the first pixel of the first row, $pv(i, 0) = pixel(i - 1, 0) \oplus pixel(i, 0). $ 
		\item 
		$pv(i, j) = \left\{ 
		\begin{array}{rl}
			max(x_2, x_3)   &, x_1 \leq min(x_2, x_3)\\
			min(x_2, x_3)   &, x_1 \geq max(x_2, x_3)\\
			x_2 + x_3 - x_1 &, otherwise
		\end{array} \right. $ 
		\item replace the $t+1$ MSB of encrypted $pixel(i, j)$ with the $t+1$ MSB of $pv(i, j)$
	\end{enumerate}
	
	\item Decrypt the cover image.
	\item[] INPUT: Encrypted image, key stream
	\item[] OUTPUT: Cover image
	
\end{enumerate}
