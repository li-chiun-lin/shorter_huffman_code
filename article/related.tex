for all $x_{i, j} \in I$, the predicted value is defined as~\eqref{eq:pv}.
The predicted error between each pixel and its predicted value is defined as~\eqref{eq:pe}.
Each pixel, except those in the first row and the first column, has a corresponding label value $t$, as defined in~\eqref{eq:t}.
All the label values forms a label map denoted as $T$.


\begin{align}
	\label{eq:pv}
	pv_{i, j} & = 
	\begin{cases}
		\text{undefined}							   & \text{if } i = 0 \vee j = 0 \\
		\max(x_{i - 1, j}, x_{i, j - 1})               & \text{if } x_{i - 1, j - 1} \leq \min(x_{i - 1, j}, x_{i, j - 1}) \\
		\min(x_{i - 1, j}, x_{i, j - 1})               & \text{if } x_{i - 1, j - 1} \geq \max(x_{i - 1, j}, x_{i, j - 1}) \\
		x_{i - 1, j} + x_{i, j - 1} - x_{i - 1, j - 1} & \textrm{otherwise} 
	\end{cases} \\
%
	\label{eq:pe}
	pe_{i, j} & = 
	\begin{cases}
		\mathrlap{\text{undefined}}
		\phantom{x_{i - 1, j} + x_{i, j - 1} - x_{i - 1, j - 1}}  & \text{if } i = 0 \vee j = 0 \\
		pv_{i, j} \oplus x_{i, j} & \text{if } 1 \leq i < h \wedge 1 \leq j < w \\
	\end{cases} \\
%
	\label{eq:t}
	t_{i, j} &= 
	\begin{dcases}
		\mathrlap{8} 
		\phantom{x_{i - 1, j} + x_{i, j - 1} - x_{i - 1, j - 1}}&  \text{if } pe_{i, j} = 0       \\
		7 - \lfloor log_2(pe_{i, j}) \rfloor &  \text{if } pe_{i, j} \neq  0
	\end{dcases}
\end{align}

The frequency of all the possible values in label map $T$ is described as~\eqref{eq:f}.

\begin{equation}
	\label{eq:f}
	F = \{f_v \mid \#(t = v), t \in T, 0\leq v \leq 8\}
\end{equation}

A predefined Huffman tree $H = \{00, 01, 100, 101, 1100, 1101, 1110, 11110, 11111\}$, the topology is visualized in~\ref{fig:fixtree}.

Let $V$ be a array consists of all the possible value of $T$ sorted by the frequency.

\begin{equation}
	V = \{v \in \range(T) \mid \forall v_i, v_j \in V \colon i \leq j \implies f_{v_i} \leq f_{v_j} \}
\end{equation}

Let $HEncoder$ be a mapping $T \mapsto H$.

\begin{align}
	& HEncoder(t) = h_c \notag \\
	& \quad	\text{ where } t \in T \wedge h_c \in H \wedge \exists c \colon v_c \in V \wedge v_c = t. 
\end{align}

\begin{equation}
	Embed(i, j, S) = 
	\begin{cases}
		MSB^{(t_{i, j} + 1)}(S) + LSB^{(7 - t_{i, j})}(x_{i, j}) & \text{if } 0 \leq t_{i, j} < 7 \\
		MSB^8(S)                                               & \text{if } 7 \leq t_{i, j} \leq  8
	\end{cases} 
\end{equation}

\begin{equation}
	Restore(i, j) = 
	\begin{cases}
		pv_{i, j}                          & \text{if } t_{i, j} = 8 \\
		\\
		\begin{aligned}
			&MSB^{t_{i, j}}(pv_{i, j}) \\
			&\quad + LSB^{(7 - t_{i, j})}(x^\prime_{i, j}) \\
			&\quad + \underbrace{(1 - {x^\prime_{i, j}}^{(7 - t_{i, j})}) \cdot 2^{(7 - t_{i, j})}}_
				{\text{flip the } (t_{i, j}+1) \text{-th bit}} 
		\end{aligned} & \text{if } 0 \leq t_{i, j} \leq 7
	\end{cases}
\end{equation}

\begin{equation}
	LEN  = \sum_{t=0}^{8} (F_t \cdot \lvert HuffEncode(t, H) \rvert)
\end{equation}
