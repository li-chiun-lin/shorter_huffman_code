\begin{equation}
	pv_{i, j} = 
	\begin{cases}
		max(x_{i - 1, j}, x_{i, j - 1})                &  \text{if } x_{i - 1, j - 1} \leq min(x_{i - 1, j}, x_{i, j - 1}), \\
		min(x_{i - 1, j}, x_{i, j - 1})                &  \text{if } x_{i - 1, j - 1} \geq max(x_{i - 1, j}, x_{i, j - 1}), \\
		x_{i - 1, j} + x_{i, j - 1} - x_{i - 1, j - 1} &  \textrm{otherwise}. 
	\end{cases} 
\end{equation}

\begin{equation}
	pe_{i, j} = 
		\begin{cases}
			\text{undefined}          & \text{if } i = 0 \vee j = 0 \\
			pv_{i, j} \oplus x_{i, j} & \text{if } 1 \leq i < h \wedge 1 \leq j < w \\
		\end{cases}
\end{equation}

\begin{equation}
	t_{i, j} = 
	\begin{dcases}
		8                                    &  \text{if } pe_{i, j} = 0 ,      \\
		7 - \lfloor log_2(pe_{i, j}) \rfloor &  \text{otherwise}.
	\end{dcases} 
\end{equation}

\begin{equation}
	Embed(i, j, S) = 
	\begin{cases}
		MSB^{(t_{i, j} + 1)}(S) + LSB^{(7 - t_{i, j})}(x_{i, j}) & \text{if } 0 \leq t_{i, j} < 7 \\
		MSB^8(S)                                               & \text{if } 7 \leq t_{i, j} \leq  8
	\end{cases} 
\end{equation}

\begin{equation}
	Restore(i, j) = 
	\begin{cases}
		pv_{i, j}                          & \text{if } t_{i, j} = 8 \\
		\\
		\begin{aligned}
			&MSB^{t_{i, j}}(pv_{i, j}) \\
			&\qquad + LSB^{(7 - t_{i, j})}(x^\prime_{i, j}) \\
			&\qquad + \underbrace{(1 - {x^\prime_{i, j}}^{(7 - t_{i, j})}) \cdot 2^{(7 - t_{i, j})}}_
				{\text{flip the } (t_{i, j}+1) \text{-th MSB}} 
		\end{aligned} & \text{if } 0 \leq t_{i, j} \leq 7
	\end{cases}
\end{equation}

\begin{equation}
	LM  = \sum_{t=0}^{8} (freq_t \cdot \lvert HuffEncode(t) \rvert)
\end{equation}


where
\begin{align}
	x_{i, j}     & \in I, \notag &\\ 
	x_{i, j}^{u} & \text{ is the } u \text{-th bit of }x_{i, j},  \notag& \\
	i            & = \{u \in \mathbb{N} \mid 1 \leq u < h\}, \notag& \\
	j            & = \{v \in \mathbb{N} \mid 1 \leq v < w\}, \notag& \\
	h            & \text{ is the height of image, }  \notag &\\ 
	w            & \text{ is the width of image, }  \notag & \\
	MSB^{m}(B)   & = \sum\nolimits_{i = 8 - m}^7 b_i \cdot 2^i, m \in [1, 8],  \notag &\\
	LSB^{n}(B)   & = \sum\nolimits_{i = 0}^{n - 1} b_i \cdot 2^i, n \in [1, 8],  \notag &\\
	B            & \in \{0, 1\}^*, b_i \in B, i \in [0, 7], \notag &\\
	S            & \in \{0, 1\}^*,  \notag  & \\
	 HuffEncode(x) &\text{ denotes the Huffman code of } x, \notag& \\ 
	 \lvert y \rvert &\text{ denotes the length of } y \notag &
\end{align}
