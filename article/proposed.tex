proposed method, ours~\cite{Grochowski2020}

three improvements:

\begin{enumerate}

	\item Reduce the size of Reference Bytes.
	\item Fix the issue of the 20-bits of length.
\end{enumerate}

Huffman codes are optimal prefix codes.


\begin{algorithm}
	\caption{Serialize a Huffman coding tree}
	
	\begin{algorithmic}[1]
		\Require $root$ : the root node of a Huffman coding tree.
		\Require $s$ : the size of bits of each value $v$.
		\Ensure $bits\_topology$ : a bit sequence represents the topology of the tree.
		\Ensure $bits\_value$ : a bit sequence represents every leaf value $v$ of the tree.
		
		\State $bits\_topology \gets \emptyset$.
		\State $bits\_value    \gets \emptyset$.
		
		\State \Call{SerializeTree}{$root$}.
		\Procedure{SerializeTree}{$root$}   
		\If{$root$ is a leaf}
		\State Let leaf value $v = (v_0, \dots, v_{s-1})$.
		\State $bits\_topology \gets bits\_topology \Vert 0$.
		\State $bits\_value \gets bits\_value \Vert v_0 \Vert \dots \Vert v_{s-1}$.
		\Else
		\State $bits\_topology \gets bits\_topology \Vert 1$.
		\State \Call{SerializeTree}{$node.left\_child$}.
		\State \Call{SerializeTree}{$node.right\_child$}.
		\EndIf
		\EndProcedure
	\end{algorithmic}
\end{algorithm}

\begin{algorithm}
	\caption{Unserialize a Huffman coding tree}
	
	\begin{algorithmic}[1]
		\Require $bits\_topology$ : a bit sequence represents the topology of the tree.
		\Require $bits\_value$ : a bit sequence represents every leaf value $v$ of the tree.
		\Require $s$ : the size of bits of each value $v$.
		\Ensure a Huffman coding tree.
		
		\Function{UnserializeTree}{$bits\_topology, bits\_value, s$}   
		\State Read 1 bit $b$ from $bits\_topology$.
%		\State Remove $b$ from $bits\_topology$.
		\If{$b = 0$ }
%		\State Let value $v$ be the first $size\_value$ bits of $bits\_value$.
%		\State Let value $v = (bits\_value_0, \dots, bits\_value_{s - 1})$
		\State Create a node $n$ with value $v = (bits\_value_0, \dots, bits\_value_{s - 1})$.
		\State Remove $s$ bits from $bits\_value$.
		
		\Else
%		\State Let $n1 \gets$ \Call{UnserializeTree}{$bits\_topology, bits\_value$}.
%		\State Let $n2 \gets$ \Call{UnserializeTree}{$bits\_topology, bits\_value$}.
		\State Create a node $n$ with
		\State\hspace{\algorithmicindent} $left\_child \gets$ \Call{UnserializeTree}{$bits\_topology, bits\_value$}.
		\State\hspace{\algorithmicindent} $right\_child \gets$ \Call{UnserializeTree}{$bits\_topology, bits\_value$}.
		\EndIf
		\State \Return $n$.
		\EndFunction
	\end{algorithmic}
\end{algorithm}

	pe(i, j) = 
	\begin{cases}
		\text{undefined},             & \text{if } i = 0 \wedge  0 \leq j < REF\_BYTES \\
		x_{i, j - 1} \oplus x_{i, j}, & \text{if } i = 0 \\
		x_{i - 1, j} \oplus x_{i, j}, & \text{if } j = 0 \\
		pv(i, j) \oplus x_{i, j}, 	  & \text{if } 1 \leq i < h \wedge 1 \leq j < w \\
	\end{cases}
\end{equation}

\begin{align*}
	REF\_BYTES &\text{ are}  \\

	& n \times \lceil log_2(max(Range(t(i, j)))) \rceil \text{ bits for Huffman coding leaf values}\\
	& length of max HuffEncoded() \\
	& a little more 
\end{align*}
