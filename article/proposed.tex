\begin{algorithm}
	\caption{Tree Serialization}
	
	\begin{algorithmic}[1]
		\Require $root$ node of a Huffman tree.
		\Require size $s$ of bits of each value $v$.
		\Ensure  bit sequence $bits\_topology$ represents the topology of the tree.
		\Ensure  bit sequence $bits\_value$ represents every leaf value $v$ of the tree.
		
		\State $bits\_topology \gets \emptyset$.
		\State $bits\_value    \gets \emptyset$.
		
		\State \Call{SerializeTree}{$root$}.
		\\
		
		\Procedure{SerializeTree}{$root$}   
		\If{$root$ is a leaf}
		\State 		Let leaf value $v$ be presented as $(v_0, \dots, v_{s-1})$.
		\State 		$bits\_topology \gets bits\_topology \Vert 0$.
		\State 		$bits\_value \gets bits\_value \Vert v_0 \Vert \dots \Vert v_{s-1}$.
		\Else
		\State		 $bits\_topology \gets bits\_topology \Vert 1$.
		\State 		\Call{SerializeTree}{$node.left\_child$}.
		\State 		\Call{SerializeTree}{$node.right\_child$}.
		\EndIf
		\EndProcedure
	\end{algorithmic}
\end{algorithm}

\begin{algorithm}
	\caption{Tree Unserialization}
	
	\begin{algorithmic}[1]
		\Require bit sequence $bits\_topology$ represents the topology of the tree.
		\Require bit sequence $bits\_value$ represents every leaf value $v$ of the tree.
		\Require size $s$ of bits of each value $v$.
		\Ensure Huffman tree $H$.
		
		\State \Call{UnserializeTree}{$bits\_topology, bits\_value$}.
		\\
		\Function{UnserializeTree}{$bits\_topology, bits\_value$}   
		\State Read 1 bit $b$ from $bits\_topology$.
		\If{$b = 0$ }
		\State 		Create a node $n$ with value $v = (bits\_value_0, \dots, bits\_value_{s - 1})$.
		\State 		Remove $s$ bits from $bits\_value$.
		
		\Else
		\State 		Create a node $n$ with
		\State\hspace{\algorithmicindent} $left\_child \gets$ \Call{UnserializeTree}{$bits\_topology, bits\_value$}.
		\State\hspace{\algorithmicindent} $right\_child \gets$ \Call{UnserializeTree}{$bits\_topology, bits\_value$}.
		\EndIf
		\State \Return $n$.
		\EndFunction
	\end{algorithmic}
\end{algorithm}
